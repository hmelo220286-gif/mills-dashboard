const SUPABASE_URL = process.env.SUPABASE_URL; const SUPABASE_KEY = process.env.SUPABASE_KEY; export default async function handler(req, res) { res.setHeader('Access-Control-Allow-Origin', '*'); res.setHeader('Access-Control-Allow-Methods', 'GET, POST, DELETE, OPTIONS'); res.setHeader('Access-Control-Allow-Headers', 'Content-Type'); if (req.method === 'OPTIONS') return res.status(200).end(); if (!SUPABASE_URL || !SUPABASE_KEY) { return res.status(500).json({ error: 'Variáveis de ambiente não configuradas' }); } const base = `${SUPABASE_URL}/rest/v1/anotacoes`; const headers = { 'apikey': SUPABASE_KEY, 'Authorization': `Bearer ${SUPABASE_KEY}`, 'Content-Type': 'application/json', 'Prefer': 'return=representation' }; try { // GET — retorna todas as anotações if (req.method === 'GET') { const r = await fetch(`${base}?select=frota,dp,ob&order=frota`, { headers }); const data = await r.json(); const result = {}; (data || []).forEach(row => { if (row.frota) result[row.frota] = { dp: row.dp || '', ob: row.ob || '' }; }); return res.status(200).json(result); } // POST — salvar/atualizar campo específico if (req.method === 'POST') { let body = req.body; if (typeof body === 'string') { try { body = JSON.parse(body); } catch(e) { body = {}; } } // frota: código do equipamento // field: 'dp' ou 'ob' — qual campo está sendo editado // value: novo valor const { frota, field, value } = body || {}; if (!frota || !field) return res.status(400).json({ error: 'frota e field obrigatórios' }); if (field !== 'dp' && field !== 'ob') return res.status(400).json({ error: 'field inválido' }); // Verificar se já existe registro para esta frota const check = await fetch( `${base}?frota=eq.${encodeURIComponent(frota)}&select=id,dp,ob`, { headers } ); const existing = await check.json(); if (existing && existing.length > 0) { // PATCH — atualizar APENAS o campo enviado const update = { updated_at: new Date().toISOString() }; update[field] = value; const r = await fetch(`${base}?frota=eq.${encodeURIComponent(frota)}`, { method: 'PATCH', headers, body: JSON.stringify(update) }); if (!r.ok) { const err = await r.text(); return res.status(r.status).json({ error: err }); } } else { // INSERT — criar nova linha com apenas o campo preenchido const insert = { frota, dp: '', ob: '', updated_at: new Date().toISOString() }; insert[field] = value; const r = await fetch(base, { method: 'POST', headers, body: JSON.stringify(insert) }); if (!r.ok) { const err = await r.text(); return res.status(r.status).json({ error: err }); } } return res.status(200).json({ ok: true }); } return res.status(405).json({ error: 'Método não permitido' }); } catch (err) { return res.status(500).json({ error: err.message }); } }
